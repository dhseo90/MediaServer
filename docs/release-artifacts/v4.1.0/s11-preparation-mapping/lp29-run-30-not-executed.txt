LP29 미실행 명령 준비 원본
소유 임시root에서 node run-30.mjs로 실행할 준비였으나 권한 심사에서 두 번 거절되어 프로세스를 생성하지 않았다.
제품/검증기의 새 공식 명령이 아닌 실행 환경·원출력 보존용 이력이다. 원래 root는 정리했으므로 다시 실행하려면 소유root 재생성과 명시 실행 승인이 필요하다.

import fs from 'node:fs';
import net from 'node:net';
import dgram from 'node:dgram';
import path from 'node:path';
import {spawn,execFileSync} from 'node:child_process';
const root='/private/tmp/media-server-lp29.dXgaDi',repo='/Users/dhseo/Workspace/mediaServer';
process.umask(0o077);
for(const d of ['output','state','events','recordings','cache','tmp'])fs.mkdirSync(path.join(root,d),{mode:0o700});
const port=()=>new Promise((resolve,reject)=>{const s=net.createServer();s.once('error',reject);s.listen(0,'127.0.0.1',()=>{const p=s.address().port;s.close(e=>e?reject(e):resolve(p));});});
const http=await port();let rtsp=await port();while(rtsp===http)rtsp=await port();
const udp=dgram.createSocket('udp4');await new Promise((resolve,reject)=>{udp.once('error',reject);udp.bind(0,'127.0.0.1',resolve);});
const env={...process.env};for(const k of Object.keys(env))if(k.startsWith('MEDIA_SERVER_')||k.startsWith('GST_')||k==='BASH_ENV')delete env[k];
Object.assign(env,{
TMPDIR:path.join(root,'tmp'),MEDIA_SERVER_SKIP_LOCAL_ENV:'1',MEDIA_SERVER_VERIFY_PREDEV_SKIP_LOCAL_ENV:'1',
MEDIA_SERVER_ENABLE_AI:'1',MEDIA_SERVER_ENABLE_YOUTUBE_SOURCE:'0',MEDIA_SERVER_BUILD_DIR:path.join(repo,'build-gst-onnx'),MEDIA_SERVER_VERIFY_PREDEV_BUILD_DIR:path.join(repo,'build-gst-onnx'),
MEDIA_SERVER_VERIFY_PREDEV_HTTP_PORT:String(http),MEDIA_SERVER_VERIFY_PREDEV_RTSP_PORT:String(rtsp),MEDIA_SERVER_VERIFY_PREDEV_HTTP_LISTEN_ADDRESS:'127.0.0.1',MEDIA_SERVER_VERIFY_PREDEV_RTSP_LISTEN_ADDRESS:'127.0.0.1',
MEDIA_SERVER_UI_BROWSER_MODE:'in-app',MEDIA_SERVER_AUTH_MODE:'off',MEDIA_SERVER_VERIFY_PREDEV_AUTH_MODE:'off',
MEDIA_SERVER_STATE_DIR:path.join(root,'state'),MEDIA_SERVER_AUTH_USERS_FILE:path.join(root,'state/users.json'),MEDIA_SERVER_SOURCE_REGISTRY:path.join(root,'state/sources.json'),MEDIA_SERVER_PUBLISHED_VIEWS:path.join(root,'state/views.json'),MEDIA_SERVER_ANALYSIS_REGISTRY:path.join(root,'state/analysis.json'),
MEDIA_SERVER_RECORDING_STORAGE_ROOT:path.join(root,'recordings'),MEDIA_SERVER_ANALYSIS_EVENT_STORAGE_PATH:path.join(root,'events/events.jsonl'),MEDIA_SERVER_ANALYSIS_EVENT_SNAPSHOT_DIR:path.join(root,'events/snapshots'),MEDIA_SERVER_ANALYSIS_EVENT_CLIP_DIR:path.join(root,'events/clips'),
MEDIA_SERVER_GST_CACHE_DIR:path.join(root,'cache'),GST_REGISTRY:path.join(root,'cache/registry.bin'),GST_REGISTRY_1_0:path.join(root,'cache/registry.bin'),
MEDIA_SERVER_WEBRTC_STUN_SERVER:`stun://127.0.0.1:${udp.address().port}`,MEDIA_SERVER_WEBRTC_TURN_SERVER:'',MEDIA_SERVER_WEBRTC_ICE_TRANSPORT_POLICY:'all'});
const args=['verify-predev','--soak-minutes','30','--skip-build','--fail-fast','--summary-file',path.join(root,'output/predev-summary.json'),'--report-file',path.join(root,'output/predev-report.md'),'--report-html-file',path.join(root,'output/predev-report.html')];
const started=process.hrtime.bigint(),meta={command:['./server.sh',...args],source:execFileSync('git',['rev-parse','HEAD'],{cwd:repo,encoding:'utf8'}).trim(),startedUtc:new Date().toISOString(),startedNs:String(started),ports:{http,rtsp,udp:udp.address().port},environment:env.MEDIA_SERVER_STATE_DIR,tokenStart:null,tokenEnd:null,tokenConsumed:null,tokenReason:'전용 집계 없음'};
fs.writeFileSync(path.join(root,'output/run-start.json'),JSON.stringify(meta,null,2)+'\n');
const fd=fs.openSync(path.join(root,'output/console.log'),'wx',0o600);
const child=spawn('./server.sh',args,{cwd:repo,env,stdio:['ignore','pipe','pipe']});
console.log(JSON.stringify({pid:child.pid,ports:meta.ports,root,started:meta.startedUtc}));
for(const stream of [child.stdout,child.stderr])stream.on('data',b=>fs.writeSync(fd,b));
child.on('error',e=>fs.writeSync(fd,JSON.stringify({spawnError:e.code||'unknown'})+'\n'));
child.on('close',(code,signal)=>{fs.closeSync(fd);udp.close();const result={exit:code,signal,endedUtc:new Date().toISOString(),endedNs:String(process.hrtime.bigint()),elapsedMs:Number(process.hrtime.bigint()-started)/1e6};fs.writeFileSync(path.join(root,'output/run-end.json'),JSON.stringify(result,null,2)+'\n');console.log(JSON.stringify(result));process.exitCode=code===0&&!signal?0:1;});
